% Gutter pull quotes follow the companion essay's composition. Native TeX
% paragraph shapes keep it compatible with the project's Tectonic bundle.
\newlength{\EssayColumnWidth}
\setlength{\EssayColumnWidth}{\dimexpr(\textwidth-\columnsep)/2\relax}
% In-flow artwork keeps each illustration beside the passage it serves.
\newcommand{\EssayArt}[2][\linewidth]{%
  \par\vspace{2mm}\noindent\makebox[\linewidth][c]{%
    \includegraphics[width=#1]{illustrations/#2.png}}\par\vspace{2mm}%
}
\newenvironment{EssayColumns}{%
  \par\noindent\begin{minipage}[t]{\EssayColumnWidth}\vspace{0pt}%
  \setlength{\parindent}{1em}\setlength{\parskip}{0pt}%
}{%
  \par\ClearGutterQuote\end{minipage}\par
}
\newcommand{\NextEssayColumn}{%
  \par\ClearGutterQuote\end{minipage}\hfill
  \begin{minipage}[t]{\EssayColumnWidth}\vspace{0pt}%
  \setlength{\parindent}{1em}\setlength{\parskip}{0pt}%
}
% Round each illustrated/header block to whole body lines. Both columns then
% share one baseline grid, even when their headings start at different heights.
\newsavebox{\GutterLeadBox}
\newcount\GutterLeadLines
\newcount\GutterLeftLeadLines
\newcommand{\GutterLead}[1]{%
  \par\setbox\GutterLeadBox=\vbox{\hsize=\EssayColumnWidth #1\par}%
  \GutterLeadLines=\number\dimexpr\ht\GutterLeadBox+\dp\GutterLeadBox+\baselineskip-1sp\relax
  \divide\GutterLeadLines by \number\baselineskip
  \nointerlineskip\vbox to\dimexpr\GutterLeadLines\baselineskip\relax{%
    \unvbox\GutterLeadBox\vfil}%
  \nointerlineskip
}
\newsavebox{\GutterQuoteBox}
\newsavebox{\GutterQuoteTextBox}
\newlength{\GutterQuoteCentre}
\newlength{\GutterQuoteTextWidth}
\newlength{\GutterTextWidth}
\newlength{\GutterRightTextWidth}
\newlength{\GutterTextIndent}
\newcount\GutterTop
\newcount\GutterBottom
\newcount\GutterWindowLines
\newcount\GutterLinesUsed
\newcount\GutterShapeLine
% Measure the actual italic lettering, including its separate opening mark.
% Keep 2.25mm clearance beside the opening mark and 4.5mm on the right.
% Centre the quote vertically in that rectangle rather than shifting one edge.
\newcommand{\PrepareGutterQuote}[2]{%
  \GutterTop=#1\relax
  \GutterLeadLines=0
  \global\GutterLeftLeadLines=0
  \setlength{\GutterQuoteCentre}{\dimexpr\EssayColumnWidth+\columnsep/2\relax}%
  \sbox{\GutterQuoteTextBox}{%
    \color{black}\fontsize{16}{20}\selectfont\itshape
    \shortstack[l]{#2}}%
  \setlength{\GutterQuoteTextWidth}{\wd\GutterQuoteTextBox}%
  \sbox{\GutterQuoteTextBox}{\vtop{%
    \hsize=\GutterQuoteTextWidth
    \color{black}\fontsize{16}{20}\selectfont\itshape\raggedright
    #2\par}}%
  \sbox{\GutterQuoteBox}{%
    \begin{tikzpicture}
      \node[anchor=north west,inner sep=0pt,outer sep=0pt] at (11mm,0)
        {\usebox{\GutterQuoteTextBox}};
      \node[anchor=north west,inner sep=0pt,outer sep=0pt,text=CoverOrange] at (0,-0.8mm)
        {\resizebox{8.5mm}{!}{\usefont{T1}{ptm}{b}{n}\textquotedblleft}};
    \end{tikzpicture}}%
  \setlength{\GutterRightTextWidth}{\dimexpr(\textwidth-\wd\GutterQuoteBox)/2-4.5mm\relax}%
  \setlength{\GutterTextWidth}{\dimexpr\GutterRightTextWidth+2.25mm\relax}%
  \GutterWindowLines=\number\dimexpr\ht\GutterQuoteBox+\dp\GutterQuoteBox+8mm+\baselineskip-1sp\relax
  \divide\GutterWindowLines by \number\baselineskip
}
% The line count spans paragraphs; the final shape line restores full width.
\newcommand{\ShapeGutterParagraph}{%
  \ifnum\GutterLinesUsed<\GutterBottom
    \GutterShapeLine=\GutterLinesUsed
    \edef\GutterShape{\number\numexpr\GutterBottom-\GutterLinesUsed+1\relax}%
    \loop\ifnum\GutterShapeLine<\GutterBottom
      \ifnum\GutterShapeLine<\GutterTop
        \edef\GutterShape{\GutterShape\space 0pt \the\EssayColumnWidth}%
      \else
        \edef\GutterShape{\GutterShape\space\the\GutterTextIndent\space\the\GutterTextWidth}%
      \fi
      \advance\GutterShapeLine by 1
    \repeat
    \edef\GutterShape{\GutterShape\space 0pt \the\EssayColumnWidth}%
    \expandafter\parshape\GutterShape\relax
  \fi
}
\newcommand{\StartGutterShape}{%
  \global\GutterLinesUsed=0
  \GutterBottom=\numexpr\GutterTop+\GutterWindowLines\relax
  \AddToHook{para/begin}[essay-gutter]{\ShapeGutterParagraph}%
  \AddToHook{para/after}[essay-gutter]{\global\advance\GutterLinesUsed by \prevgraf}%
}
\newcommand{\ClearGutterQuote}{%
  \RemoveFromHook{para/begin}[essay-gutter]%
  \RemoveFromHook{para/after}[essay-gutter]%
  \parshape=0
}
\newcommand{\GutterQuote}{%
  \global\GutterLeftLeadLines=\GutterLeadLines
  \setlength{\GutterTextIndent}{0pt}%
  \StartGutterShape
  \noindent\begin{tikzpicture}[overlay]
    \node[anchor=center,inner sep=0pt,outer sep=0pt]
      at (\GutterQuoteCentre,\dimexpr-\GutterTop\baselineskip-\GutterWindowLines\baselineskip/2+\ht\strutbox\relax)
      {\usebox{\GutterQuoteBox}};
  \end{tikzpicture}\ignorespaces
}
% Derive the mirrored cutout from the measured header heights. No independent
% right-column vertical offset can skew its upper and lower boundaries.
\newcommand{\ReserveGutterQuote}{%
  \advance\GutterTop by \GutterLeftLeadLines
  \advance\GutterTop by -\GutterLeadLines
  \setlength{\GutterTextWidth}{\GutterRightTextWidth}%
  \setlength{\GutterTextIndent}{\dimexpr\EssayColumnWidth-\GutterTextWidth\relax}%
  \StartGutterShape
  \ignorespaces
}
