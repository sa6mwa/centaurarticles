% Page composition is explicit: column breaks occur only between paragraphs.
% Each column pair is indivisible, so an overfull page fails verification rather
% than silently cutting a paragraph or letting a figure drift into another section.
% wrapstuff v0.3 disables emergency stretch when it rebreaks the narrow lines
% beside a quote. A small fallback lets TeX choose a legal earlier word break
% without opening up the spacing excessively. Fail if the package changes.
\RequirePackage{etoolbox}
\ExplSyntaxOn
\patchcmd{\__wstf_build_window:}
  {\dim_zero:N \tex_emergencystretch:D}
  {\dim_set:Nn \tex_emergencystretch:D {1em}}{}
  {\PackageError{coordination-trap}{Cannot~preserve~wrapped-text~line~breaking}
    {Review~the~wrapstuff~adapter~in~editorial-layout.tex.}}
\ExplSyntaxOff

\newlength{\EssayColumnWidth}
\setlength{\EssayColumnWidth}{\dimexpr(\textwidth-\columnsep)/2\relax}
\newenvironment{EssayColumns}{%
  \par\noindent\begin{minipage}[t]{\EssayColumnWidth}\vspace{0pt}%
  \setlength{\parindent}{1em}\noindent
}{%
  \par\end{minipage}\par
}
\newcommand{\NextEssayColumn}{%
  \par\end{minipage}\hfill
  \begin{minipage}[t]{\EssayColumnWidth}\vspace{0pt}%
  \setlength{\parindent}{1em}\noindent
}
\newcommand{\EssayArt}[1]{%
  \par\vspace{8mm}\noindent\includegraphics[width=\textwidth]{#1}\par
}
% Measure the lettering itself, then wrap both columns around the same box.
% The mirrored phantom reserves precisely the same space without painting twice.
\newsavebox{\GutterQuoteBox}
\newlength{\GutterQuoteOverhang}
\newcommand{\PrepareGutterQuote}[3]{%
  \def\GutterQuoteTop{#1}%
  \setlength{\GutterQuoteOverhang}{\dimexpr(#2+\columnsep)/2\relax}%
  \sbox{\GutterQuoteBox}{\parbox{#2}{%
    \centering\color{black}\fontsize{17}{20}\selectfont
    {\color{black!45}\textquotedblleft}#3{\color{black!45}\textquotedblright}\par}}%
}
\newcommand{\GutterQuote}{%
  \begin{wrapstuff}[r,top=\GutterQuoteTop,hoffset=\GutterQuoteOverhang,
      leftsep=4.5mm,vsep=1.5mm]
    \usebox{\GutterQuoteBox}%
  \end{wrapstuff}%
}
\newcommand{\NextGutterColumn}{%
  \par\wrapstuffclear\NextEssayColumn
  \begin{wrapstuff}[l,top=\GutterQuoteTop,hoffset=\GutterQuoteOverhang,
      rightsep=4.5mm,vsep=1.5mm]
    \phantom{\usebox{\GutterQuoteBox}}%
  \end{wrapstuff}%
}
