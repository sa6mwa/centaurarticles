% Page composition is explicit: column breaks occur only between paragraphs.
% Each column pair is indivisible, so an overfull page fails verification rather
% than silently cutting a paragraph or letting a figure drift into another section.
\newlength{\EssayColumnWidth}
\setlength{\EssayColumnWidth}{\dimexpr(\textwidth-\columnsep)/2\relax}
\newenvironment{EssayColumns}{%
  \par\noindent\begin{minipage}[t]{\EssayColumnWidth}\vspace{0pt}%
  \setlength{\parindent}{1em}\noindent
}{%
  \par\end{minipage}\par
}
\newcommand{\NextEssayColumn}{%
  \par\end{minipage}\hfill
  \begin{minipage}[t]{\EssayColumnWidth}\vspace{0pt}%
  \setlength{\parindent}{1em}\noindent
}
\newcommand{\EssayArt}[1]{%
  \par\vspace{8mm}\noindent\includegraphics[width=\textwidth]{#1}\par
}
\newcommand{\ContinuedSection}[1]{%
  \clearpage\noindent{\color{blue}\sffamily\small\bfseries #1 / continued}\par\vspace{6mm}
}

% One short pull quote straddles the gutter. Matching wrapstuff windows narrow
% the facing edges of the two columns for the same six lines; the prose then
% returns to its normal width. The quote is painted once, from the left column.
\newcommand{\GutterQuote}{%
  \begin{wrapstuff}[r,top=14,width=70mm,height=25mm,hoffset=38.5mm,
      leftsep=7mm,lines=6]
    \centering\color{black}\fontsize{18}{21}\selectfont
    {\color{black!45}\textquotedblleft}The mandate must\\move real decisions.{\color{black!45}\textquotedblright}\par
  \end{wrapstuff}%
}
\newcommand{\NextGutterColumn}{%
  \par\wrapstuffclear\NextEssayColumn
  \begin{wrapstuff}[l,top=14,width=70mm,height=25mm,hoffset=38.5mm,
      rightsep=7mm,lines=6]
    \mbox{}%
  \end{wrapstuff}%
}
